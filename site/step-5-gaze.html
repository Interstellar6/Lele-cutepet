第 3 版 · 盯着你

目标：眼睛跟着鼠标转。原理还是动同一层贴片 —— 这次是 朝鼠标方向平移几个像素，空出来的一侧露出皮上的眼窝。

<script>
(function () {
  'use strict';

  const SPRITE = { w: 811, h: 792 };
  const W = 220;
  const H = Math.round(W * SPRITE.h / SPRITE.w);
  const PAD_X = Math.round(W * 0.20);
  const PAD_T = Math.round(H * 0.32);
  const PAD_B = Math.round(H * 0.16);
  const CW = W + PAD_X * 2;
  const CH = PAD_T + H + PAD_B;
  const DPR = Math.min(2, window.devicePixelRatio || 1);
  const K = W / SPRITE.w;

  const EYES = {
    l: { box: [154, 156, 362, 352] },
    r: { box: [521, 176, 709, 360] }
  };

  const canvas = document.getElementById('c');
  const ctx = canvas.getContext('2d');
  canvas.width = CW * DPR;
  canvas.height = CH * DPR;
  canvas.style.width = CW + 'px';
  canvas.style.height = CH + 'px';

  const stage = document.getElementById('stage');
  const hit = document.getElementById('hit');
  const dot = document.getElementById('dot');
  const floorY = () => window.innerHeight * 0.78 - H;

  const pet = {
    x: window.innerWidth * 0.72 - W / 2,
    y: floorY(),
    squash: null,
    facing: 1                      // ← 第 3 步新增：朝左还是朝右
  };

  function render() {
    stage.style.transform =
      'translate3d(' + Math.round(pet.x - PAD_X) + 'px,' +
                       Math.round(pet.y - PAD_T) + 'px,0)';
  }

  hit.style.left = (PAD_X + W * 0.12) + 'px';
  hit.style.right = (PAD_X + W * 0.12) + 'px';
  hit.style.top = (PAD_T + H * 0.18) + 'px';
  hit.style.bottom = PAD_B + 'px';

  let skin = null, eyeL = null, eyeR = null;
  Promise.all([
    '../assets/lele-skin.png', '../assets/lele-eye-l.png', '../assets/lele-eye-r.png'
  ].map(loadImage)).then((arr) => {
    skin = arr[0]; eyeL = arr[1]; eyeR = arr[2];
    layout();
    requestAnimationFrame(frame);
  });
  function loadImage(src) {
    return new Promise((res, rej) => {
      const i = new Image();
      i.onload = () => res(i);
      i.onerror = rej;
      i.src = src;
    });
  }

  hit.addEventListener('pointerdown', (e) => {
    e.preventDefault();
    pet.squash = { t: 0, dur: 0.45, amp: 0.22 };
  });
  /* 开局和窗口变化时都重新贴地。
     注意：别信「解析 HTML 那一刻」的窗口尺寸 —— 它可能还不是最终尺寸，
     所以图片加载完要再 layout() 一次。 */
  function layout() {
    pet.y = floorY();
    pet.x = Math.min(pet.x, window.innerWidth - W);
    render();
  }
  addEventListener('resize', layout);

  const blink = { t: 0, dur: 0, next: 1.5 + Math.random() * 3 };
  const openEl = document.getElementById('open');
  let manual = false;
  openEl.addEventListener('input', () => { manual = true; });

  function updateBlink(dt) {
    if (manual) return +openEl.value / 100;
    if (blink.dur > 0) {
      blink.t += dt;
      const k = blink.t / blink.dur;
      if (k >= 1) { blink.dur = 0; blink.next = 1.6 + Math.random() * 4; return 1; }
      return k < 0.3 ? 1 - k / 0.3 : (k - 0.3) / 0.7;
    }
    blink.next -= dt;
    if (blink.next <= 0) { blink.dur = 0.32; blink.t = 0; }
    return 1;
  }

  /* ─────────── 第 3 步新增：鼠标 → 视线 ───────────
     目标值 gaze.tx / gaze.ty，当前值 gaze.x / gaze.y 用缓动追过去 */
  const pointer = { x: 0, y: 0, seen: false };
  const gaze = { x: 0, y: 0, tx: 0, ty: 0 };
  const flipEl = document.getElementById('flip');
  const bugEl = document.getElementById('bug');

  addEventListener('pointermove', (e) => {
    pointer.x = e.clientX;
    pointer.y = e.clientY;
    pointer.seen = true;
    dot.style.left = e.clientX + 'px';
    dot.style.top = e.clientY + 'px';
  });

  function updateGaze(dt) {
    if (pointer.seen) {
      // 两眼中间大概在「身高 1/3」的位置
      const midX = pet.x + W / 2;
      const midY = pet.y + H * 0.34;
      const dx = pointer.x - midX;
      const dy = pointer.y - midY;
      const d = Math.hypot(dx, dy) || 1;

      const reach = Math.min(1, Math.max(0.25, d / 240));   // 越远看得越用力
      gaze.tx = (dx / d) * 15 * reach;
      gaze.ty = (dy / d) * 10 * reach;
    }
    // 缓动：每帧追 10% 左右，别直接赋值（否则眼球会「咔」地跳）
    const rate = Math.min(1, dt * 10);
    gaze.x += (gaze.tx - gaze.x) * rate;
    gaze.y += (gaze.ty - gaze.y) * rate;
  }

  let last = performance.now();
  let time = 0;

  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    time += dt;

    if (pet.squash) {
      pet.squash.t += dt;
      if (pet.squash.t >= pet.squash.dur) pet.squash = null;
    }

    pet.facing = flipEl.checked ? -1 : 1;
    updateGaze(dt);
    draw(updateBlink(dt));
    requestAnimationFrame(frame);
  }

  function draw(open) {
    let sy = 1 + Math.sin(time * 2.1) * 0.012;
    let sx = 1 - Math.sin(time * 2.1) * 0.008;
    let bob = -Math.sin(time * 2.1) * 1.4;

    if (pet.squash) {
      const k = pet.squash.t / pet.squash.dur;
      const s = Math.sin(k * Math.PI) * (1 - k * 0.5) * pet.squash.amp * 1.25;
      sy *= 1 - s;
      sx *= 1 + s * 0.9;
    }

    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
    ctx.clearRect(0, 0, CW, CH);
    shadow();

    ctx.save();
    ctx.translate(PAD_X + W / 2, PAD_T + H);
    ctx.scale(sx * pet.facing, sy);        // ← 镜像在这里发生
    ctx.translate(-W / 2, -H + bob);

    ctx.drawImage(skin, 0, 0, W, H);
    drawEye(eyeL, EYES.l.box, open);
    drawEye(eyeR, EYES.r.box, open);

    ctx.restore();
  }

  function drawEye(img, box, open) {
    const bx = box[0] * K;
    const by = box[1] * K;
    const bw = (box[2] - box[0]) * K;
    const bh = (box[3] - box[1]) * K;

    const dh = bh * Math.max(0.06, Math.min(1, open));

    /* ─────────── 第 3 步新增：横向偏移要乘 facing ───────────
       身体镜像后，本地坐标的 +x 在屏幕上其实是往左。
       不乘 facing 的话，猫一朝左，眼球就跟着反了。 */
    const gx = bugEl.checked ? gaze.x : gaze.x * pet.facing;

    ctx.drawImage(img, bx + gx * K, by + gaze.y * K + (bh - dh) / 2, bw, dh);
  }

  function shadow() {
    const ox = PAD_X + W / 2;
    const oy = PAD_T + H + PAD_B * 0.34;
    const rx = W * 0.42;
    const ry = Math.max(3, PAD_B * 0.18);
    const g = ctx.createRadialGradient(ox, oy, 1, ox, oy, rx);
    g.addColorStop(0, 'rgba(70,60,95,.30)');
    g.addColorStop(1, 'rgba(70,60,95,0)');
    ctx.save();
    ctx.translate(ox, oy);
    ctx.scale(1, ry / rx);
    ctx.translate(-ox, -oy);
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.arc(ox, oy, rx, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }
})();
</script>

新增：一个方向 + 一次缓动

搜 updateGaze。核心就三行：算出鼠标相对「两眼中间」的单位方向，乘上最大偏移量：

const reach = Math.min(1, Math.max(0.25, d / 240));   // 越远「看得越用力」
gaze.tx = (dx / d) * 15 * reach;                      // 水平最多 15 个精灵像素
gaze.ty = (dy / d) * 10 * reach;                      // 垂直少一点，更自然

然后不要直接赋值，用缓动追过去： gaze.x += (gaze.tx - gaze.x) * Math.min(1, dt * 10)。 直接赋值的话鼠标一动眼球就「咔」地跳一下，很出戏。

新增：镜像的坑（重点看这里）

这一版加了 pet.facing：朝左时用 ctx.scale(-1, 1) 把整只猫翻个面。 问题来了 —— 镜像之后，本地坐标的 +x 在屏幕上其实是往左。 

const gx = bugEl.checked ? gaze.x : gaze.x * pet.facing;
ctx.drawImage(img, bx + gx * K, by + gaze.y * K + (bh - dh) / 2, bw, dh);
⛔
这个 bug 为什么难发现 它只在猫朝左时才反。而「朝向」通常还会被存进 localStorage， 于是它往左走过一次之后，你下次打开页面看到的就一直是反的。 勾上演示里的两个开关，自己看一眼就懂了。
