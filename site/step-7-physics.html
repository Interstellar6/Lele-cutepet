<script>
(function () {
  'use strict';

  const SPRITE = { w: 811, h: 792 };
  const W = 220;
  const H = Math.round(W * SPRITE.h / SPRITE.w);
  const PAD_X = Math.round(W * 0.20);
  const PAD_T = Math.round(H * 0.32);
  const PAD_B = Math.round(H * 0.16);
  const CW = W + PAD_X * 2;
  const CH = PAD_T + H + PAD_B;
  const DPR = Math.min(2, window.devicePixelRatio || 1);
  const K = W / SPRITE.w;

  const EYES = {
    l: { box: [154, 156, 362, 352] },
    r: { box: [521, 176, 709, 360] }
  };

  /* ─────────── 第 4 步新增：腮红位置 ───────────
     精灵坐标，落在脸颊上。改图之后这两个数要跟着调。 */
  const CHEEKS = [{ x: 186, y: 356, r: 96 }, { x: 672, y: 372, r: 92 }];

  /* ─────────── 第 4 步新增：台词 ─────────── */
  const LINES = {
    happy: ['嘿嘿～', '好开心！', '喵呜～'],
    shy:   ['别、别盯着看…', '讨厌啦', '脸有点热…'],
    angry: ['哼！不理你了', '再点试试看', '喵喵喵！'],
    pat:   ['呼噜呼噜…', '好舒服～', '再摸一下下']
  };
  const pick = (arr) => arr[Math.floor(Math.random() * arr.length)];

  /* ─────────── 第 4 步新增：表情 ───────────
     每个表情就是一个函数：拿到当前姿态 p 和进度 k（0~1），随便改。
     好处：表情可以叠加（开心 + 眨眼 + 呼吸同时成立）。 */
  const EMOTES = {
    joy: { dur: 1.1, line: 'happy', apply(p, k) {
      const w = Math.sin(k * Math.PI * 3) * (1 - k);     // (1-k) 让尾巴自然衰减
      p.sy *= 1 + 0.09 * w;
      p.sx *= 1 - 0.05 * w;
      p.rot += Math.sin(k * Math.PI * 4) * 0.05 * (1 - k);
      p.blush = Math.max(p.blush, 0.75 * Math.sin(k * Math.PI));
      p.bob -= Math.abs(Math.sin(k * Math.PI)) * 6;
    }},
    shy: { dur: 1.4, line: 'shy', apply(p, k) {
      const a = Math.sin(Math.min(1, k * 1.6) * Math.PI) * (k > 0.75 ? (1 - k) / 0.25 : 1);
      p.blush = Math.max(p.blush, 0.95 * a);
      p.rot += 0.12 * a;
      p.gazeX -= 9 * a;                                   // 害羞：眼睛往旁边躲
    }},
    angry: { dur: 1.0, line: 'angry', apply(p, k) {
      p.rot += Math.sin(k * Math.PI * 12) * 0.05 * (1 - k);
      p.sx *= 1 + 0.04 * Math.sin(k * Math.PI * 6) * (1 - k);
      p.eyeOpen = Math.min(p.eyeOpen, 0.75);
    }}
  };

  const canvas = document.getElementById('c');
  const ctx = canvas.getContext('2d');
  canvas.width = CW * DPR;
  canvas.height = CH * DPR;
  canvas.style.width = CW + 'px';
  canvas.style.height = CH + 'px';

  const stage = document.getElementById('stage');
  const hit = document.getElementById('hit');
  const bubbleEl = document.getElementById('bubble');
  const floorY = () => window.innerHeight * 0.78 - H;

  /* ─────────── 第 5 步新增：状态机 ───────────
     idle（站着）→ drag（被拎着）→ fly（飞/下落）→ idle
     只要这三个状态，别写更多。 */
  const clamp = (v, a, b) => (v < a ? a : v > b ? b : v);

  const pet = {
    x: window.innerWidth * 0.72 - W / 2,
    y: floorY(),
    squash: null,
    facing: 1,
    state: 'idle',        // idle | drag | fly
    vx: 0, vy: 0,
    rot: 0, rotV: 0,
    target: null,         // 拖动时指针相对角色的偏移
    walking: 0,
    walkPhase: 0
  };

  /* ─────────── 第 4 步新增：表情 / 腮红 / 粒子状态 ─────────── */
  let emote = null;        // { name, t }
  let blush = 0;           // 0~1
  const parts = [];        // 爱心粒子

  function render() {
    stage.style.transform =
      'translate3d(' + Math.round(pet.x - PAD_X) + 'px,' +
                       Math.round(pet.y - PAD_T) + 'px,0)';
  }

  hit.style.left = (PAD_X + W * 0.12) + 'px';
  hit.style.right = (PAD_X + W * 0.12) + 'px';
  hit.style.top = (PAD_T + H * 0.18) + 'px';
  hit.style.bottom = PAD_B + 'px';

  /* 开局和窗口变化时都重新贴地。
     注意：别信「解析 HTML 那一刻」的窗口尺寸 —— 它可能还不是最终尺寸，
     所以图片加载完要再 layout() 一次。 */
  function layout() {
    pet.y = floorY();
    pet.x = Math.min(pet.x, window.innerWidth - W);
    render();
  }
  addEventListener('resize', layout);

  let skin = null, eyeL = null, eyeR = null;
  Promise.all([
    '../assets/lele-skin.png', '../assets/lele-eye-l.png', '../assets/lele-eye-r.png'
  ].map(loadImage)).then((arr) => {
    skin = arr[0]; eyeL = arr[1]; eyeR = arr[2];
    layout();
    requestAnimationFrame(frame);
  });
  function loadImage(src) {
    return new Promise((res, rej) => {
      const i = new Image();
      i.onload = () => res(i);
      i.onerror = rej;
      i.src = src;
    });
  }

  const blink = { t: 0, dur: 0, next: 1.5 + Math.random() * 3 };

  function updateBlink(dt) {
    if (blink.dur > 0) {
      blink.t += dt;
      const k = blink.t / blink.dur;
      if (k >= 1) { blink.dur = 0; blink.next = 1.6 + Math.random() * 4; return 1; }
      return k < 0.3 ? 1 - k / 0.3 : (k - 0.3) / 0.7;
    }
    blink.next -= dt;
    if (blink.next <= 0) { blink.dur = 0.32; blink.t = 0; }
    return 1;
  }

  const pointer = { x: 0, y: 0, seen: false };
  const gaze = { x: 0, y: 0, tx: 0, ty: 0 };

  /* ─────────── 第 4 步新增：抚摸判定 ───────────
     数「横向来回变向」的次数：3 个来回就呼噜。
     pat.cool 是冷却，必须每帧递减，否则点一次之后永久失效。 */
  const pat = { dir: 0, count: 0, cool: 0 };

  addEventListener('pointermove', (e) => {
    pointer.x = e.clientX;
    pointer.y = e.clientY;
    pointer.seen = true;
    detectPat(e);
  });

  function detectPat(e) {
    if (pet.state === 'drag') return;          // 拖拽中不判抚摸
    if (pat.cool > 0) { pat.dir = 0; return; }
    const r = stage.getBoundingClientRect();
    const inside = e.clientX > r.left - 12 && e.clientX < r.right + 12 &&
                   e.clientY > r.top - 12 && e.clientY < r.bottom + 12;
    if (!inside) { pat.dir = 0; return; }

    if (pointer.lx == null) { pointer.lx = e.clientX; return; }
    const dx = e.clientX - pointer.lx;
    pointer.lx = e.clientX;
    if (Math.abs(dx) < 2) return;

    const dir = dx > 0 ? 1 : -1;
    if (pat.dir && dir !== pat.dir && ++pat.count >= 3) {
      pat.count = 0;
      purr();
      return;
    }
    pat.dir = dir;
  }

  function purr() {
    pat.cool = 2.5;
    blush = 1;
    emote = { name: 'joy', t: 0 };
    spawn(5);
    say(pick(LINES.pat));
  }

  /* ─────────── 第 4 步：点一下 → 随机表情 ─────────── */
  let lastClickAt = 0;
  let clicks = 0;

  function poke() {
    const now = performance.now();
    clicks = now - lastClickAt < 1600 ? clicks + 1 : 1;   // 连点 5 次炸毛
    lastClickAt = now;

    const name = clicks >= 5 ? 'angry' : pick(['joy', 'shy', 'angry']);
    emote = { name, t: 0 };
    blush = Math.max(blush, 0.5);
    if (name === 'joy') spawn(3);

    const e2 = EMOTES[name];
    if (e2.line && Math.random() < 0.8) say(pick(LINES[e2.line]));
  }

  /* ─────────── 第 5 步新增：拖拽 + 甩出 ───────────
     按下记一个起点；移动超过 7px 才算「拖」；
     松手时用最近 120ms 的采样点估速度，够快就飞出去。 */
  let pid = null, downAt = 0, downX = 0, downY = 0, moved = false;
  const samples = [];

  hit.addEventListener('pointerdown', (e) => {
    pid = e.pointerId;
    try { hit.setPointerCapture(pid); } catch (_) {}
    downAt = performance.now();
    downX = e.clientX;
    downY = e.clientY;
    samples.length = 0;
    samples.push({ t: downAt, x: e.clientX, y: e.clientY });
    moved = false;
  });

  hit.addEventListener('pointermove', (e) => {
    if (pid === null || e.pointerId !== pid) return;

    const now = performance.now();
    samples.push({ t: now, x: e.clientX, y: e.clientY });
    while (samples.length > 2 && now - samples[0].t > 120) samples.shift();

    if (!moved) {
      if (Math.hypot(e.clientX - downX, e.clientY - downY) <= 7) return;
      moved = true;
      pet.state = 'drag';
      stage.classList.add('dragging');
    }

    /* 抓点比中心偏下一点：像拎着后颈，而不是顶着天灵盖 */
    pet.target = { x: e.clientX - W / 2, y: e.clientY - H * 0.62 };
    e.preventDefault();
  }, { passive: false });

  function release(e) {
    if (pid === null || e.pointerId !== pid) return;
    pid = null;
    stage.classList.remove('dragging');

    const wasDrag = moved;
    moved = false;

    if (!wasDrag) { poke(); return; }      // 没拖动 = 点了一下

    const s0 = samples[0];
    const dt0 = Math.max(16, performance.now() - s0.t) / 1000;
    const vx = (e.clientX - s0.x) / dt0;
    const vy = (e.clientY - s0.y) / dt0;

    pet.target = null;
    pet.state = 'fly';
    if (Math.hypot(vx, vy) > 620) {         // 够快才叫「甩」
      pet.vx = vx * 0.85;
      pet.vy = vy * 0.85 - 120;             // 稍微往上抛一点，更好看
      pet.rotV = clamp(vx / 900, -6, 6);
    } else {
      pet.vx = vx * 0.25;
      pet.vy = vy * 0.25;
    }
  }

  hit.addEventListener('pointerup', release);
  hit.addEventListener('pointercancel', release);

  /* ─────────── 第 4 步新增：气泡 ─────────── */
  let sayTimer = 0;
  function say(text) {
    bubbleEl.textContent = text;
    bubbleEl.classList.add('on');
    clearTimeout(sayTimer);
    sayTimer = setTimeout(() => bubbleEl.classList.remove('on'), 2600);
  }

  /* ─────────── 第 4 步新增：爱心粒子 ─────────── */
  function spawn(n) {
    for (let i = 0; i < n; i++) {
      const life = 0.9 + Math.random() * 0.6;
      parts.push({
        x: PAD_X + W * (0.25 + Math.random() * 0.5),
        y: PAD_T + H * 0.2,
        vx: (Math.random() - 0.5) * 70,
        vy: -60 - Math.random() * 70,
        life, max: life,
        s: 0.7 + Math.random() * 0.6,
        rot: 0,
        rv: (Math.random() - 0.5) * 4
      });
    }
  }

  function updateGaze(dt) {
    if (pointer.seen) {
      const midX = pet.x + W / 2;
      const midY = pet.y + H * 0.34;
      const dx = pointer.x - midX;
      const dy = pointer.y - midY;
      const d = Math.hypot(dx, dy) || 1;
      const reach = Math.min(1, Math.max(0.25, d / 240));
      gaze.tx = (dx / d) * 15 * reach;
      gaze.ty = (dy / d) * 10 * reach;
      if (pet.state === 'drag') { gaze.tx *= 0.5; gaze.ty = -10; }   // 被拎着：往上看
    }
    const rate = Math.min(1, dt * 10);
    gaze.x += (gaze.tx - gaze.x) * rate;
    gaze.y += (gaze.ty - gaze.y) * rate;
  }

  /* ─────────── 第 5 步新增：物理 ───────────
     重力、撞地反弹、撞墙反弹、落地挤压；闲着的时候自己走回原位 */
  const G = 2600;              // 重力加速度（px/s²），凭手感调
  const BOUNCE = 0.42;         // 撞地损失多少速度
  const FRICTION = 0.72;       // 地面摩擦
  const WALL = 0.55;           // 撞墙损失
  const LAND_MIN = 240;        // 落地竖速低于这个就直接站住

  function updatePhysics(dt) {
    const floor = floorY();

    if (pet.state === 'drag') {
      /* 弹簧跟随：每帧朝目标追 25% 左右，别瞬间贴住指针 */
      const k = Math.min(1, dt * 16);
      const px = pet.x;
      const tx = pet.target ? pet.target.x : pet.x;
      const ty = pet.target ? pet.target.y : pet.y;
      pet.x += (tx - pet.x) * k;
      pet.y += (ty - pet.y) * k;

      pet.vx = (pet.x - px) / Math.max(dt, 0.0001);
      /* 随横向速度轻轻歪头 */
      pet.rot += (-clamp(pet.vx / 2600, -0.3, 0.3) - pet.rot) * Math.min(1, dt * 8);
      return;
    }

    if (pet.state === 'fly') {
      pet.vy += G * dt;
      pet.x += pet.vx * dt;
      pet.y += pet.vy * dt;
      pet.rot += pet.rotV * dt;
      pet.rotV *= (1 - Math.min(1, dt * 1.6));

      if (pet.y >= floor) {
        pet.y = floor;
        if (Math.abs(pet.vy) > LAND_MIN) {
          pet.vy = -pet.vy * BOUNCE;
          pet.vx *= FRICTION;
          pet.rotV += (Math.random() - 0.5) * 2;
          const amp = clamp(Math.abs(pet.vy) / 1600, 0.16, 0.44);
          pet.squash = { t: 0, dur: 0.34, amp };     // ← 落地挤压，重量感全在这
        } else {
          land();
        }
      }
      if (pet.y < -H * 0.5) { pet.y = -H * 0.5; pet.vy = Math.abs(pet.vy) * 0.5; }
      if (pet.x < 2) { pet.x = 2; pet.vx = Math.abs(pet.vx) * WALL; }
      if (pet.x > window.innerWidth - W - 2) {
        pet.x = window.innerWidth - W - 2;
        pet.vx = -Math.abs(pet.vx) * WALL;
      }
      if (Math.abs(pet.vx) > 60) pet.facing = pet.vx > 0 ? 1 : -1;
      return;
    }

    /* idle：回正、慢慢走回家 */
    pet.rot += (0 - pet.rot) * Math.min(1, dt * 6);
    pet.vx = 0;
    pet.vy = 0;

    const home = window.innerWidth * 0.72 - W / 2;
    if (Math.abs(pet.x - home) > 8) {
      pet.walking = pet.x < home ? 1 : -1;
      pet.x += pet.walking * 90 * dt;
      pet.facing = pet.walking;
      pet.walkPhase += dt * 9;
    } else {
      pet.walking = 0;
    }
  }

  function land() {
    pet.state = 'idle';
    pet.vx = 0;
    pet.vy = 0;
    pet.rotV = 0;
    pet.squash = { t: 0, dur: 0.42, amp: 0.26 };
  }

  let last = performance.now();
  let time = 0;

  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    time += dt;

    if (pet.squash) {
      pet.squash.t += dt;
      if (pet.squash.t >= pet.squash.dur) pet.squash = null;
    }
    if (emote) {
      emote.t += dt;
      if (emote.t >= EMOTES[emote.name].dur) emote = null;
    }
    if (pat.cool > 0) pat.cool = Math.max(0, pat.cool - dt);
    blush = Math.max(0, blush - dt * 0.35);

    updatePhysics(dt);
    updateGaze(dt);
    draw(updateBlink(dt), dt);
    render();                       /* 位置变了要重新摆 DOM */
    requestAnimationFrame(frame);
  }

  /* ─────────── 第 4 步新增：把「这一帧的姿态」抽出来 ───────────
     所有表情、呼吸、挤压都只改这个对象，最后统一交给 draw 去画。
     这样加表情不用碰绘制代码。 */
  function computePose(open) {
    const p = {
      rot: 0,
      sx: 1 - Math.sin(time * 2.1) * 0.008,
      sy: 1 + Math.sin(time * 2.1) * 0.012,
      bob: -Math.sin(time * 2.1) * 1.4,
      gazeX: gaze.x,
      gazeY: gaze.y,
      eyeOpen: open,
      blush
    };

    if (pet.squash) {
      const k = pet.squash.t / pet.squash.dur;
      const s = Math.sin(k * Math.PI) * (1 - k * 0.5) * pet.squash.amp * 1.25;
      p.sy *= 1 - s;
      p.sx *= 1 + s * 0.9;
    }

    if (emote) {
      const e = EMOTES[emote.name];
      e.apply(p, Math.min(1, emote.t / e.dur));
    }

    /* ─────────── 第 5 步新增：被动姿势 ─────────── */
    if (pet.state === 'drag') {                 // 被拎着：随速度拉长
      const st = Math.min(0.25, Math.abs(pet.vx) / 1400);
      p.sy *= 1 + st;
      p.sx *= 1 - st * 0.6;
    }
    if (pet.walking) {                          // 走路：上下颠（没有腿也能看出在走）
      const w = Math.abs(Math.sin(pet.walkPhase));
      p.bob -= w * H * 0.045;
      p.sy *= 1 + w * 0.03;
    }
    p.rot += pet.rot;                           // 飞行/倾斜的整体旋转

    p.eyeOpen = Math.max(0.06, Math.min(1, p.eyeOpen));
    return p;
  }

  function draw(open, dt) {
    const p = computePose(open);

    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
    ctx.clearRect(0, 0, CW, CH);
    shadow();

    ctx.save();
    ctx.translate(PAD_X + W / 2, PAD_T + H);
    ctx.rotate(p.rot);
    ctx.scale(p.sx * pet.facing, p.sy);
    ctx.translate(-W / 2, -H + p.bob);

    ctx.drawImage(skin, 0, 0, W, H);
    drawEye(eyeL, EYES.l.box, p);
    drawEye(eyeR, EYES.r.box, p);

    /* ─────────── 第 4 步新增：腮红 ───────────
       source-atop = 只画在「已经有像素」的地方，也就是猫身上，
       所以腮红不会糊到画布空白处，也不用手动抠形状。 */
    if (p.blush > 0.02) {
      ctx.globalCompositeOperation = 'source-atop';
      for (const c of CHEEKS) {
        const cx = c.x * K;
        const cy = c.y * K;
        const r = c.r * K;
        const g = ctx.createRadialGradient(cx, cy, r * 0.1, cx, cy, r);
        g.addColorStop(0, 'rgba(255,120,155,' + (0.55 * p.blush).toFixed(3) + ')');
        g.addColorStop(1, 'rgba(255,160,185,0)');
        ctx.fillStyle = g;
        ctx.beginPath();
        ctx.ellipse(cx, cy, r, r * 0.72, 0, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.globalCompositeOperation = 'source-over';
    }

    ctx.restore();

    drawParticles(dt);
  }

  function drawEye(img, box, p) {
    const bx = box[0] * K;
    const by = box[1] * K;
    const bw = (box[2] - box[0]) * K;
    const bh = (box[3] - box[1]) * K;
    const dh = bh * p.eyeOpen;
    const gx = p.gazeX * pet.facing;
    ctx.drawImage(img, bx + gx * K, by + p.gazeY * K + (bh - dh) / 2, bw, dh);
  }

  /* ─────────── 第 4 步新增：画爱心 ───────────
     画在身体变换之外，所以不受挤压影响 */
  function drawParticles(dt) {
    for (let i = parts.length - 1; i >= 0; i--) {
      const q = parts[i];
      q.life -= dt;
      q.x += q.vx * dt;
      q.y += q.vy * dt;
      q.vy += 130 * dt;
      q.rot += q.rv * dt;
      if (q.life <= 0) { parts.splice(i, 1); continue; }

      ctx.save();
      ctx.globalAlpha = Math.max(0, q.life / q.max);
      ctx.translate(q.x, q.y);
      ctx.rotate(q.rot);
      ctx.scale(q.s, q.s);
      ctx.fillStyle = '#ff6b95';
      ctx.beginPath();
      ctx.moveTo(0, 4);
      ctx.bezierCurveTo(-7, -2, -5, -9, 0, -5);
      ctx.bezierCurveTo(5, -9, 7, -2, 0, 4);
      ctx.fill();
      ctx.restore();
    }
  }

  function shadow() {
    const ox = PAD_X + W / 2;
    const oy = PAD_T + H + PAD_B * 0.34;
    const rx = W * 0.42;
    const ry = Math.max(3, PAD_B * 0.18);
    const g = ctx.createRadialGradient(ox, oy, 1, ox, oy, rx);
    g.addColorStop(0, 'rgba(70,60,95,.30)');
    g.addColorStop(1, 'rgba(70,60,95,0)');
    ctx.save();
    ctx.translate(ox, oy);
    ctx.scale(1, ry / rx);
    ctx.translate(-ox, -oy);
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.arc(ox, oy, rx, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }
})();

新增：拖拽用弹簧，不要跟手

const k = Math.min(1, dt * 16);       // 每帧朝目标追 25% 左右
pet.x += (tx - pet.x) * k;
pet.y += (ty - pet.y) * k;

直接 pet.x = e.clientX 会像鼠标拖着一张图片，很「死」。 再配上「身体随速度拉长」，才像拎着一只活物： 

const st = Math.min(0.25, Math.abs(pet.vx) / 1400);
p.sy *= 1 + st;      // 纵向拉长
p.sx *= 1 - st * 0.6; // 横向变细

新增：用最近 120ms 估速度

// 每次 pointermove 记一个采样点，只留最近 120ms
samples.push({ t: now, x: e.clientX, y: e.clientY });
while (samples.length > 2 && now - samples[0].t > 120) samples.shift();

// 松手时算速度
const dt0 = Math.max(16, now - samples[0].t) / 1000;
const vx = (e.clientX - samples[0].x) / dt0;
if (Math.hypot(vx, vy) > 620) { /* 甩出去 */ }

窗口取太短会抖，太长会把「停下再松手」误判成快甩。120ms 是个好起点。

新增：弹跳与落地挤压

if (pet.y >= floor) {
  pet.y = floor;
  if (Math.abs(pet.vy) > 240) {
    pet.vy = -pet.vy * 0.42;        // 撞地反弹，一次比一次矮
    pet.vx *= 0.72;                 // 地面摩擦
    pet.squash = { t: 0, dur: 0.34, amp };   // ← 这一下最有重量感
  } else {
    land();                         // 速度小了就站住
  }
}
常量取值调大 / 调小会怎样
G2600调大更「重」，调小像月球
BOUNCE0.42调大弹更多次
FRICTION0.72调小落地滑得更远
LAND_MIN240低于这个直接站住、不再弹
甩出阈值620调小更容易甩飞
